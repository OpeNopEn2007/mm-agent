  import subprocess

  def execute_code_with_retry(code_path: str, task_id: str, max_execute: int = 5, max_repair: int = 3, timeout: int = 300) -> dict:
      """Execute generated Python code with timeout and error repair."""

      result = {
          "task_id": task_id,
          "status": "failed",
          "execution_time": 0,
          "stdout": "",
          "stderr": "",
          "results": {},
          "plots": [],
          "created_at": datetime.now().isoformat()
      }

      execute_attempts = 0
      repair_attempts = 0

      while execute_attempts < max_execute:
          execute_attempts += 1

          try:
              import time
              start_time = time.time()

              # Execute code with timeout (Decision D-07)
              process = subprocess.run(
                  ["python3", code_path],
                  capture_output=True,
                  text=True,
                  timeout=timeout,
                  cwd=".planning"
              )

              execution_time = time.time() - start_time
              result["execution_time"] = execution_time
              result["stdout"] = process.stdout
              result["stderr"] = process.stderr

              if process.returncode == 0:
                  result["status"] = "success"
                  # Parse stdout for results JSON
                  try:
                      # Look for JSON output in stdout
                      stdout_lines = process.stdout.strip().split('\n')
                      for line in stdout_lines:
                          try:
                              parsed = json.loads(line)
                              if "numerical_solution" in parsed or "fitting_parameters" in parsed:
                                  result["results"] = parsed
                                  break
                          except json.JSONDecodeError:
                              continue
                  except Exception as e:
                      result["results"] = {"error": str(e)}

                  # Check for generated plots (Decision D-10)
                  import os
                  plot_dir = f".planning/output/plots/{task_id}"
                  if os.path.exists(plot_dir):
                      for plot_file in os.listdir(plot_dir):
                          if plot_file.endswith('.png') or plot_file.endswith('.pdf'):
                              result["plots"].append({
                                  "path": os.path.join(plot_dir, plot_file),
                                  "type": plot_file.split('.')[0],
                                  "description": f"Plot: {plot_file}"
                              })

                  return result  # Success
              else:
                  # Non-zero exit code - error
                  result["stderr"] = process.stderr
                  raise subprocess.CalledProcessError(process.returncode, code_path, process.stdout, process.stderr)

          except subprocess.TimeoutExpired as e:
              # Timeout handling (Decision D-07)
              e.kill()  # Terminate process
              e.wait()  # Reap zombie process
              result["status"] = "timeout"
              result["stderr"] = f"Execution exceeded {timeout}s timeout"

              # First timeout: try simplifying code
              if execute_attempts == 1:
                  # LLM repair for timeout: simplify computation
                  repair_code = llm_repair_timeout(code_path, timeout)
                  if repair_code:
                      with open(code_path, 'w') as f:
                          f.write(repair_code)
                      repair_attempts += 1
                      continue  # Retry with simplified code
                  else:
                      return result  # Timeout after failed repair
              else:
                  return result  # Timeout on retry

          except Exception as e:
              # Other errors - LLM repair (Decision D-08)
              result["stderr"] = str(e)
              error_class = classify_error(e)

              if repair_attempts < max_repair and error_class in ["syntax", "runtime", "import"]:
                  # Repair attempt
                  repaired_code = llm_repair_code(
                      original_code=code_path,
                      traceback=str(e),
                      error_class=error_class
                  )
                  if repaired_code:
                      # Validate repaired code with AST
                      try:
                          ast.parse(repaired_code)
                          with open(code_path, 'w') as f:
                              f.write(repaired_code)
                          repair_attempts += 1
                          continue  # Retry with repaired code
                      except SyntaxError:
                          # Repair failed validation
                          pass

              # Max retries exhausted or unfixable error
              return result

      result["status"] = "failed"
      result["stderr"] = f"Max execution attempts ({max_execute}) exhausted"
      return result

  def classify_error(exception: Exception) -> str:
      """Classify error type for targeted repair (Decision D-08)."""
