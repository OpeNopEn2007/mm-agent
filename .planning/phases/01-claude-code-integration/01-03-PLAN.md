    # Validate required parameter
    if [[ -z "$PROBLEM_FILE" ]]; then
      echo "Error: --problem parameter is required"
      echo "Usage: /mm-agent --problem <file> [--interactive] [--skip-verify] [--phase N]"
      exit 1
    fi

    echo "PROBLEM_FILE=$PROBLEM_FILE"

    if [[ ! "$EXT" =~ ^(pdf|md|txt)$ ]]; then
      echo "Error: Unsupported file format: .$EXT"
      echo "Supported formats: PDF, Markdown (.md), Plain text (.txt)"
