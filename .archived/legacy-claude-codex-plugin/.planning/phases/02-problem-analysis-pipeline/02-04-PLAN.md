    Create PDF versions of test fixtures using Pandoc for smoke testing PROB-01.

    For simple.pdf:
    ```bash
    # Check if pandoc is installed
    if ! command -v pandoc &> /dev/null; then
      echo "Error: pandoc not installed. Installing..."
      # macOS
      if [[ "$OSTYPE" == "darwin"* ]]; then
        brew install pandoc
      # Linux
      else
        sudo apt-get install pandoc
      fi
    fi

    # Convert simple.md to simple.pdf
    pandoc tests/fixtures/simple.md -o tests/fixtures/simple.pdf \
      --pdf-engine=xelatex \
      --variable geometry:margin=1in \
      --variable fontsize=12pt

    # Verify PDF was created
    test -f tests/fixtures/simple.pdf || exit 1

    echo "✓ Created simple.pdf"
    ```

    For multi-task.pdf:
    ```bash
    # Convert multi-task.md to multi-task.pdf
    pandoc tests/fixtures/multi-task.md -o tests/fixtures/multi-task.pdf \
      --pdf-engine=xelatex \
      --variable geometry:margin=1in \
      --variable fontsize=12pt

    # Verify PDF was created
    test -f tests/fixtures/multi-task.pdf || exit 1

    echo "✓ Created multi-task.pdf"
    ```

    If Pandoc is not available, create placeholder PDFs with a note:
    ```bash
    # Create placeholder simple.pdf (only if Pandoc failed)
    if [ ! -f "tests/fixtures/simple.pdf" ]; then
      echo "Note: Pandoc not available, creating placeholder PDF"
      # Use Python to create minimal PDF
      python3 << 'EOF'
      from reportlab.lib.pagesizes import letter
      from reportlab.pdfgen import canvas
      from reportlab.lib.units import inch

      c = canvas.Canvas("tests/fixtures/simple.pdf", pagesize=letter)

      # Read markdown content
      with open("tests/fixtures/simple.md", "r") as f:
          text = f.read()

      # Write text to PDF (simple line-by-line)
      y = letter[1] - inch
      for line in text.split('\n'):
          c.drawString(inch, y, line)
          y -= 14  # 14pt line spacing
          if y < inch:
              c.showPage()
              y = letter[1] - inch

      c.save()
      EOF

      echo "✓ Created placeholder simple.pdf (minimal formatting)"
    fi
    ```
