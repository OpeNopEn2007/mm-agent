   def _analyze_dependencies_heuristic(tasks_data):
       """
       Simple heuristic for dependency analysis based on keywords.

       For v1, uses simple pattern matching. Future: LLM-based analysis.
       """
       dependency_keywords = {
           'based on': 1, 'using': 1, 'depends on': 1,
           'after': 1, 'requires': 1, 'given': 1
       }

       task_ids = sorted(tasks_data.keys(), key=int)
       dependencies = {}

       for i, task_id in enumerate(task_ids):
           desc = tasks_data[task_id]['description'].lower()
           deps = []

           # Check if this task references previous tasks
           for prev_id in task_ids[:i]:
               prev_desc = tasks_data[prev_id]['description'].lower()
               # If current task mentions previous task keywords
               if any(keyword in desc for keyword in dependency_keywords):
                   # Simple heuristic: depends on immediate previous task
                   if i > 0 and task_ids[i-1] not in deps:
                       deps.append(task_ids[i-1])

           dependencies[task_id] = deps

       # Update tasks_data
       for task_id, deps in dependencies.items():
           tasks_data[task_id]['dependencies'] = deps

       return tasks_data
   ```

   from graphlib import TopologicalSorter, CycleError

   def topological_sort(tasks_data):
       """
       Perform topological sort on DAG and detect cycles.

       Args:
           tasks_data: Dict of task_id -> {description, dependencies, status}

       Returns:
           Tuple: (execution_order, cycle_info)
                  - execution_order: List of task IDs in order, or None if cycle
                  - cycle_info: Dict with 'has_cycle' and 'cycle' path, or None

       Raises:
           SystemExit: On cycle detection (with formatted error message)
       """
       # Build graph for TopologicalSorter
       graph = {
           task_id: set(task_info['dependencies'])
           for task_id, task_info in tasks_data.items()
       }

       sorter = TopologicalSorter(graph)

       # Try to prepare - this detects cycles
       try:
           sorter.prepare()
       except CycleError as e:
           cycle_path = list(e.args[1])  # List of nodes forming the cycle
           error_msg = format_cycle_error(cycle_path, tasks_data)
           print(error_msg, file=sys.stderr)
           return None, {'has_cycle': True, 'cycle': cycle_path}

       # Get execution order
       execution_order = list(sorter.static_order())

       # Update tasks_data with execution order
       tasks_data['execution_order'] = execution_order

       return execution_order, None
   ```

   def format_cycle_error(cycle_path, tasks_data):
       """
       Format detailed error message for circular dependency (Decision D-02).
       """
       cycle_str = " → ".join(cycle_path + [cycle_path[0]])  # Close the loop

       deps_detail = []
       for task_id in cycle_path:
           task = tasks_data.get(task_id, {})
           deps = task.get('dependencies', [])
           deps_detail.append(f"- Task {task_id}: depends on {deps}")

       return f"""❌ 循环依赖检测失败

循环链: {cycle_str}

任务依赖详情:
{chr(10).join(deps_detail)}

修复建议:
1. 检查任务描述是否有逻辑矛盾
