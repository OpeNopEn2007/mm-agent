# Memory Schema (IDEA.md §7.2)

       """
       Load Memory files for all dependencies of a task (MEM-01).

       Args:
           task_id: Current task ID (string, e.g., "3")
           dag_path: Path to dag.json file
           memory_dir: Path to memory directory

       Returns:
           Formatted context string from all dependency memories

       Raises:
           FileNotFoundError: If dag.json or dependency Memory files not found
           KeyError: If task_id not found in DAG
       """

2. Implement DAG loading and dependency extraction:
   ```python
   import json
   from pathlib import Path

   # Load DAG
   dag_path = Path(dag_path)
   if not dag_path.exists():
       raise FileNotFoundError(f"DAG file not found: {dag_path}")

   with open(dag_path, 'r', encoding='utf-8') as f:
       dag = json.load(f)

   # Get dependencies for task_id
   if task_id not in dag['tasks']:
       raise KeyError(f"Task {task_id} not found in DAG")

   dependencies = dag['tasks'][task_id]['dependencies']
   memory_dir = Path(memory_dir)

   # Load context from each dependency
   context_parts = []
   for dep_id in dependencies:
       memory_path = memory_dir / f'task-{dep_id}.json'

       if not memory_path.exists():
           raise FileNotFoundError(f"Memory file not found: {memory_path}")

       with open(memory_path, 'r', encoding='utf-8') as f:
           memory = json.load(f)

       # Validate required fields
       required_fields = ['task_id', 'task_description', 'status']
       for field in required_fields:
           if field not in memory:
               raise ValueError(f"Memory file {memory_path} missing required field: {field}")

       # Format context (IDEA.md §7.3)
       context_parts.append(f"""
---
# Task {dep_id}: {memory['task_description']}

## Modeling Method
{memory.get('mathematical_modeling_process', 'N/A')}

## Result Interpretation
{memory.get('solution_interpretation', 'N/A')}

## Code Outputs
{json.dumps(memory.get('code_structure', {}).get('file_outputs', []), indent=2)}
---
""")
   ```

3. Return formatted context:
   ```python
   if not context_parts:
       return "# No dependencies for this task"

   return '\n'.join(context_parts)
   ```

4. Update main() function to call load_dependencies():
   ```python
   def main():
       parser = argparse.ArgumentParser(
           description='Load dependency Memory files for task context'
       )
       parser.add_argument('--task-id', required=True,
                          help='Task ID to load dependencies for')
       parser.add_argument('--dag', required=True,
                          help='Path to dag.json')
       parser.add_argument('--memory-dir', required=True,
                          help='Path to memory directory')
       parser.add_argument('--output', required=True,
                          help='Output context file path')
       args = parser.parse_args()

       try:
           context = load_dependencies(
               task_id=args.task_id,
               dag_path=args.dag,
               memory_dir=args.memory_dir
           )

           # Write context to output file
           output_path = Path(args.output)
           output_path.parent.mkdir(parents=True, exist_ok=True)

           with open(output_path, 'w', encoding='utf-8') as f:
               f.write(context)

           print(f"SUCCESS: Context written to {output_path}")
           print(f"Dependencies loaded: {len(args.task_id)}")

       except Exception as e:
           print(f"ERROR: {e}", file=sys.stderr)
           sys.exit(1)

   if __name__ == '__main__':
       main()
   ```

   def update_task_completion(task_id, result, memory_dir='.planning/memory'):
       """
       Update Memory file when task completes.

       Args:
           task_id: Task ID
           result: Dict with interpretation and optional code_structure, charts
           memory_dir: Path to memory directory

       Returns:
           Path to updated Memory file
       """
       memory_path = Path(memory_dir) / f'task-{task_id}.json'

       with open(memory_path, 'r', encoding='utf-8') as f:
           memory_data = json.load(f)

       memory_data['status'] = 'completed'
       memory_data['solution_interpretation'] = result.get('interpretation', '')
       memory_data['updated_at'] = datetime.now().isoformat()

       if 'task_code' in result:
           memory_data['task_code'] = result['task_code']
       if 'execution_result' in result:
           memory_data['execution_result'] = result['execution_result']

       return write_memory(task_id, memory_data, memory_dir)
